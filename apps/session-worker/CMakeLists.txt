cmake_minimum_required(VERSION 3.20)
project(vc_workspace_session_worker C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FREERDP REQUIRED IMPORTED_TARGET freerdp3>=3.31.0 winpr3>=3.31.0)
find_package(Threads REQUIRED)
add_library(worker-protocol STATIC protocol.c)
target_compile_options(worker-protocol PRIVATE -Wall -Wextra -Werror)
add_executable(vc-workspace-session-worker main.c)
target_compile_options(vc-workspace-session-worker PRIVATE -Wall -Wextra -Werror)
target_link_libraries(vc-workspace-session-worker PRIVATE worker-protocol PkgConfig::FREERDP Threads::Threads)
install(TARGETS vc-workspace-session-worker RUNTIME DESTINATION libexec)
include(CTest)
if(BUILD_TESTING)
  add_executable(worker-protocol-test tests/protocol.c)
  target_link_libraries(worker-protocol-test PRIVATE worker-protocol)
  target_compile_options(worker-protocol-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME worker-protocol COMMAND worker-protocol-test)
endif()
